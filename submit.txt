        public static DataTable ConvertJArrayToDataTable(JArray jArray)
        {
            DataTable dt = new DataTable();

            if (jArray == null || !jArray.Any())
                return dt;

            // 1. 根据第一个元素动态创建列
            var firstObj = jArray[0] as JObject;
            if (firstObj != null)
            {
                foreach (var prop in firstObj.Properties())
                {
                    dt.Columns.Add(prop.Name, typeof(string)); // 为了通用，全部设为 string；可根据实际类型调整
                }
            }

            // 2. 填充数据行
            foreach (var item in jArray)
            {
                var obj = item as JObject;
                if (obj == null) continue;

                DataRow row = dt.NewRow();
                foreach (DataColumn col in dt.Columns)
                {
                    JToken token = obj[col.ColumnName];
                    row[col.ColumnName] = (token == null) ? "" : token.ToString();
                }
                dt.Rows.Add(row);
            }

            return dt;
        }

        private string GetDataFromAWX_ID(Dictionary<string, string> Data, string strID, ref JObject awxData)
        {
            string res = "";
            try
            {
                res = awx.GetDataFromAWX_ID(Data, strID);
                awxData = JObject.Parse(res);
            }
            catch (System.Exception ex)
            {
                Alert(ex.Message);
            }
            return res;
        }

        /// <summary>
        /// 設置表單參數
        /// </summary>
        private void SetParameter()
        {
            formCode = Request.QueryString["formcode"] ?? "";
            var status = Request.QueryString["status"] ?? "";
            FormStatus = string.IsNullOrEmpty(status) ? FormState.Create :
                "0".Equals(status) ? FormState.Draft :
                "1".Equals(status) ? FormState.Send :
                "2".Equals(status) ? FormState.Receive : FormState.Other;
        }

        /// <summary>
        /// 保存/提交
        /// </summary>
        /// <param name="status"></param>
        /// <returns></returns>
        private bool Save(int status)
        {
            TempModel model = GetData(status);
            db = new DBClass(true);
            try
            {
                var result = db.InsertNpiProposalForm(model);
                if (result.IsSuccess)
                {
                    if (result.OutVal != null && result.OutVal.Count > 0 && result.OutVal[0] == "0")
                    {
                        db.Commit();
                        return true;
                    }
                }
            }
            catch (Exception ex)
            {
                throw; // 保留堆栈跟踪
            }
            finally { db.Close(); }
            return false;
        }

                    Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('Attachment format must be pptx！')</script>");
                    return;
                }
                string selectedId = DropDownList3.SelectedValue;
                if (string.IsNullOrEmpty(selectedId))
                {
                    Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('Category must be selected！')</script>");
                    return;
                }
                //判斷上傳文件是否符合要求
                if ("491553".Equals(selectedId))
                {
                    if (!"NpiLabProposalTemplate.pptx".Equals(fullname))
                    {
                        Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('The attached file does not meet the requirements！')</script>");
                        return;
                    }
                }
                else if ("491554".Equals(selectedId))
                {
                    if (!"NpiDeviceProposalTemplate.pptx".Equals(fullname))
                    {
                        Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('The attached file does not meet the requirements！')</script>");
                        return;
                    }
                }
                else if ("491555".Equals(selectedId))
                {
                    if (!"NpiSoftwareProposalTemplate.pptx".Equals(fullname))
                    {
                        Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('The attached file does not meet the requirements！')</script>");
                        return;
                    }
                }
                else if ("491556".Equals(selectedId))
                {
                    if (!"NpiLabProposalTemplate.pptx".Equals(fullname))
                    {
                        Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('The attached file does not meet the requirements！')</script>");
                        return;
                    }
                }
                // 保存上传的文件到服务器
               

                string year = DateTime.Now.Year.ToString();
                string month = DateTime.Now.Month.ToString();

                upfile.createFolder(Server.MapPath("~\\UploadFiles\\" + year + month));
                String strFolder = "UploadFiles\\" + year + month + "\\";
                string filetype = System.IO.Path.GetExtension(fullname);
                String strFileName = Server.MapPath("~\\") + strFolder + Guid.NewGuid().ToString("n") + filetype;

                string strSource = strFileName;
                FindFile2.PostedFile.SaveAs(strSource);

                bool isOK = sftp.TransferFile(strSource, out strFileURL, out strUpName);
                if (!isOK)
                {
                    Page.ClientScript.RegisterClientScriptBlock(this.GetType(), "alert", "<script>alert('File upload failed！')</script>");
                    return;
                }
                else {
                    //暫存文件信息
                    ViewState["FILE_PATH"] = strFileURL;
                    ViewState["FILE_NAME"] = strUpName + filetype;

                    fileNameTd.InnerText = fullname;
                    fileNameTd.Attributes["title"] = fullname;
                }
               
            }
            else
            {
                Alert(alert.ResourceManager.GetString("Common_Choose_Attah"));
            }
        }

        protected void txt_Dri_ENo_TextChanged(object sender, EventArgs e)
        {
            var empno = txt_Dri_Emp.Text.Trim();
            if (!string.IsNullOrEmpty(empno))
            {
                var result = db.GetEmpBaseInfo(new TempModel() { Property01 = empno });
                if (result.dt != null && result.dt.Rows.Count > 0)
                {

            {
                txt_Dri_Email.Text = "";
                txt_Dri_Emp.Text = "";
            }
        }
