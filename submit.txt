    for (const item of menu) {
      if (item.Id === itemId) {
        // If the current item matches the clicked item, return the hierarchy
        return [...hierarchy, itemId];
      } else if (item.children && item.children.length > 0) {
        // If the current item has children, recursively search for the clicked item in its children
        const childHierarchy = findItemHierarchy(item.children, itemId, [...hierarchy, item.Id]);
        if (childHierarchy.length > hierarchy.length) {
          // If the child hierarchy is longer, it means the clicked item is found in the children
          return childHierarchy;
        }
      }
    }
    // Return an empty array if the clicked item is not found in the current menu branch
    return [];

        return prevOpen.filter((id) => id !== itemId);
      } else {
        // If the clicked item is closed, find its hierarchy using the recursive function
        const itemHierarchy = findItemHierarchy(menu, itemId);

        // Close other open items that are not in the new hierarchy
        const filteredOpen = [...prevOpen.filter((id) => itemHierarchy.includes(id)), ...itemHierarchy];
        //console.log(filteredOpen)
        return filteredOpen;
