SELECT
    wrtv.FORM_ID,
    wrtv.FORM_CODE,
    wrtv.ROUTE_ORDER,
    wrtv.STEP_ID,
    NVL(wrtv.OPT_EMP_NAME, wfocv.APPROVE_EMP_NAME) AS OPT_EMP_NAME,
    wrtv.IS_FINISHED,
    wrtv.LAST_EDIT_BY,
    wrtv.LAST_EDIT_DT
FROM wf_structure.WF_ROUTE_TRACKING_V wrtv
LEFT JOIN wf_structure.WF_FORM_OPT_CONDITION_V wfocv
       ON wrtv.FORM_CODE = wfocv.FORM_CODE
      AND wrtv.STEP_ID = wfocv.STEP_ID
ORDER BY wrtv.FORM_ID, wrtv.ROUTE_ORDER;



steps.forEach((step: any, idx: number) => {

  if (idx >= reachedCount) {
    return;
  }

  const empNames = parseEmpNames(
    step.OPT_EMP_NAME || step.APPROVE_EMP_NAME
  );

  empNames.forEach((empName) => {

    const status = Number(step.IS_FINISHED);

    if (status === 1) {
      addToGrouped(empName, 'completed');
    }
    else if (status === -1) {
      addToGrouped(empName, 'signing');
    }

  });

});



private readonly ISM_STEP_IDS = [183, 201, 202, 203, 204, 205];



steps = steps.filter((s: any) =>
  this.ISM_STEP_IDS.includes(Number(s.STEP_ID))
);
