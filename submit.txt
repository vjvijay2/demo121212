
import { Component, OnInit } from '@angular/core';
import { HttpClient } from '@angular/common/http';

interface EmployeeStatusRow {
  name: string;
  signing: number;
  completed: number;
  rejected: number;
}

@Component({
  selector: 'app-employee-status',
  templateUrl: './employee-status.component.html',
  styleUrls: ['./employee-status.component.less']
})
export class EmployeeStatusComponent implements OnInit {

  private readonly PROPOSALS_URL = 'http://localhost:5665/api/proposals';

  // 👉 STEP_ID values நீங்க later correct பண்ணிக்கோங்க
  // First / Second level உங்களுக்கு வேண்டாம், அதனால் அவை இங்க வைக்க வேண்டியதில்லை
  private readonly STEP_ORDER = [183, 201, 202, 203, 204, 205, 206, 207, 208];

  // 👉 ISM Engineer 1 → ISM Engineer 2 மட்டும் consider பண்ண
  // இந்த list-ல உங்க actual STEP_ID values வச்சு update பண்ணணும்
  private readonly ISM_STEP_IDS = [203, 204, 205, 206, 207, 208];

  loading = true;
  error = false;
  employees: EmployeeStatusRow[] = [];
  chartOption: any = {};

  constructor(private http: HttpClient) {}

  ngOnInit(): void {
    this.fetchEmployeeStatus();
  }

  fetchEmployeeStatus(): void {

    this.loading = true;
    this.error = false;

    this.http.get<any>(this.PROPOSALS_URL).subscribe({

      next: (res: any) => {

        console.log('RAW RESPONSE', res);

        const formList: any[] =
          (res && res.List) ||
          (res && res.data) ||
          (Array.isArray(res) ? res : []);

        const stepList: any[] =
          (res && res.StepList) ||
          (res && res.Steps) ||
          [];

        console.log('Form count:', formList.length);
        console.log('Step count:', stepList.length);

        // FormStatue lookup - FORM_ID வெச்சு
        const formStatusById: { [formId: string]: string } = {};
        formList.forEach((f: any) => {
          const fid =
            f.Id != null
              ? String(f.Id)
              : f.FormId != null
                ? String(f.FormId)
                : f.FORM_ID != null
                  ? String(f.FORM_ID)
                  : '';

          if (fid) {
            formStatusById[fid] = (f.FormStatue || '').toLowerCase();
          }
        });

        // Step rows-ஐ FORM_ID வெச்சு group பண்றோம்
        const stepsByForm: { [formId: string]: any[] } = {};
        stepList.forEach((s: any) => {
          const fid = s.FORM_ID != null ? String(s.FORM_ID) : '';
          if (!fid) return;
          if (!stepsByForm[fid]) stepsByForm[fid] = [];
          stepsByForm[fid].push(s);
        });

        console.log('Forms with steps:', Object.keys(stepsByForm).length);

        const grouped: { [key: string]: EmployeeStatusRow } = {};

        const addToGrouped = (name: string, bucket: 'signing' | 'completed' | 'rejected') => {
          if (!name) return;
          if (!grouped[name]) {
            grouped[name] = { name, signing: 0, completed: 0, rejected: 0 };
          }
          grouped[name][bucket]++;
        };

        // ஒரு step-ல multiple employees இருந்தா split பண்ண (comma-separated)
        const parseEmpNames = (raw: string): string[] =>
          (raw || '')
            .split(',')
            .map(x => x.trim())
            .filter(x => x !== '');

        Object.keys(stepsByForm).forEach((formId) => {

          let steps = stepsByForm[formId];
          const overallStatus = formStatusById[formId] || '';

          console.log('--------------------------');
          console.log('FORM', formId, 'TOTAL STEPS:', steps.length);
          console.log('ALL STEP_IDs:', steps.map(s => s.STEP_ID));

          // 👉 ISM Engineer 1 → ISM Engineer 2 உள்ள steps மட்டும் எடுத்துக்கிறோம்
          steps = steps.filter((s: any) => {
            const sid = Number(s.STEP_ID);
            return this.ISM_STEP_IDS.indexOf(sid) !== -1;
          });

          console.log('ISM STEP_IDs LEFT:', steps.map(s => s.STEP_ID));

          if (!steps.length) {
            console.log('→ NO ISM STEPS, SKIPPING FORM', formId);
            return;
          }

          // STEP_ORDER வரிசைப்படி sort பண்றோம்
          steps.sort((a, b) => {
            const ai = this.STEP_ORDER.indexOf(Number(a.STEP_ID));
            const bi = this.STEP_ORDER.indexOf(Number(b.STEP_ID));
            return ai - bi;
          });

          console.log('Sorted ISM STEP_IDs:', steps.map(s => s.STEP_ID));

          // Form இப்போ எந்த ISM step-ல pending-ஆ இருக்குனு கண்டுபிடிக்க
          let currentStepIndex = steps.findIndex((s: any) => Number(s.IS_FINISHED) !== 1);

          const reachedCount = currentStepIndex === -1 ? steps.length : currentStepIndex + 1;

          console.log('currentStepIndex:', currentStepIndex, 'reachedCount:', reachedCount, 'overallStatus:', overallStatus);

          steps.forEach((step: any, idx: number) => {

            if (idx >= reachedCount) {
              console.log('Skipping future ISM step', step.STEP_ID);
              return;
            }

            const empNames = parseEmpNames(step.OPT_EMP_NAME);
            const isFinished = Number(step.IS_FINISHED) === 1;

            console.log(
              'Step', step.STEP_ID,
              'IS_FINISHED:', step.IS_FINISHED,
              'Employees:', empNames
            );

            empNames.forEach((empName) => {
              if (isFinished) {
                if (idx === reachedCount - 1 && currentStepIndex === -1 && overallStatus === 'rejected') {
                  addToGrouped(empName, 'rejected');
                  console.log('→ REJECTED count for', empName);
                } else {
                  addToGrouped(empName, 'completed');
                  console.log('→ COMPLETED count for', empName);
                }
              } else {
                addToGrouped(empName, 'signing');
                console.log('→ SIGNING count for', empName);
              }
            });

          });

        });

        this.employees = Object.values(grouped)
          .sort((a, b) => (b.signing + b.completed + b.rejected) - (a.signing + a.completed + a.rejected));

        console.log('Final employees list:', this.employees);

        this.loading = false;
      },

      error: (err) => {
        console.error(err);
        this.loading = false;
        this.error = true;
      }

    });

  }

}
