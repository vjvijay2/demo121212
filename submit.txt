import { Component, OnInit } from '@angular/core';
import { Location } from '@angular/common';
import { HttpClient } from '@angular/common/http';

interface ProposalRow {
  item: number;
  formCode: string;
  formStatus: string;
  empNo: string;
  empName: string;
  Factory: string;
  labName: string;
  category: string;
  salaryGrade: string;
  managementTitle: string;
  DeptName: string;
}

@Component({
  selector: 'app-all-proposals',
  templateUrl: './all-proposals.component.html',
  styleUrls: ['./all-proposals.component.less'],
})
export class AllProposalsComponent implements OnInit {
  private readonly TEST_URL = 'http://10.208.193.125:8089/mock/trueExam.do?id=ffff-1783663681896-10208193125-1822';
  private readonly LIVE_URL = 'https://microservice.abgind.foxconn.com/india/gateway';
  private readonly API_ID = 'ffff-1783663681896-10208193125-1822';
  private readonly USER_KEY = 'E9785F13CCEAFB251A6AB147AFD24336';

  loading = true;
  error = false;

  proposals: ProposalRow[] = [];
  filteredProposals: ProposalRow[] = [];

  currentPage = 1;  
  pageSize = 10;

  filters = {
    formNo: '',
    empId: '',
    name: '',
    dateFrom: null as Date | null,
    dateTo: null as Date | null,
    formStatus: 'ALL',
    Factory: 'ALL',
    category: 'ALL',
  };

  formStatusOptions = ['ALL', 'Completed', 'Signing', 'Pending', 'Rejected', 'Closed'];
  FactoryOptions = ['ALL', 'FXCN', 'FXBL'];
  categoryOptions = ['ALL', 'NPI Lab Proposal', 'NPI Device Proposal', 'NPI Software Proposal', 'Other'];

  constructor(private location: Location, private http: HttpClient) { }

  ngOnInit(): void {
    this.fetchProposals();
  }

  private formatDate(date: Date | null): string {
    if (!date) return '';
    const day = ('0' + date.getDate()).slice(-2);
    const month = ('0' + (date.getMonth() + 1)).slice(-2);
    const year = date.getFullYear();
    return day + '-' + month + '-' + year;
  }

  fetchProposals(): void {
    this.loading = true;
    this.error = false;

    const url = this.TEST_URL;

    this.http.get<any>(url, {
      headers: {
        apiID: this.API_ID,
        userKey: this.USER_KEY,
      },
    }).subscribe({
      next: (res: any) => {
        const list: any[] =
          (res && res.List) ||
          (res && res.data) ||
          (Array.isArray(res) ? res : []);

        this.proposals = list.map((item, idx) => ({
          item: idx + 1,
          formCode: item.FormCode || item.formCode || '',
          formStatus: item.FormStatue || item.FormStatus || item.formStatus || '',
          empNo: item.EmpNo || item.empNo || '',
          empName: item.EmpName || item.empName || '',
          Factory: item.Factory || item.factory || '',
          labName: item.LabName || item.labName || '',
         category: item.Category1 || item.Category || item.category || '',
          salaryGrade: item.SalaryGrade || item.salaryGrade || '',
          managementTitle: item.ManagementTitle || item.managementTitle || '',
          DeptName: item.DeptName || item.deptName || item.Dept || item.dept || '',
        }));

        this.applyFilters();
        this.loading = false;
      },
      error: (err) => {
        console.error('NPI Proposal list API error:', err);
        this.error = true;
        this.loading = false;
      },
    });
   console.log(this.proposals.map(x => x.formStatus));
  }



  applyFilters(): void {
    const formNo = (this.filters.formNo || '').toLowerCase();
    const empId = (this.filters.empId || '').toLowerCase();
    const name = (this.filters.name || '').toLowerCase();

    this.filteredProposals = this.proposals.filter((row) => {
      const matchFormNo = !formNo || (row.formCode || '').toLowerCase().indexOf(formNo) !== -1;
      const matchEmpId = !empId || (row.empNo || '').toLowerCase().indexOf(empId) !== -1;
      const matchName = !name || (row.empName || '').toLowerCase().indexOf(name) !== -1;

      const matchStatus =
        this.filters.formStatus === 'ALL' ||
        (row.formStatus || '').toLowerCase() === this.filters.formStatus.toLowerCase();

      const matchFactory =
        this.filters.Factory === 'ALL' ||
        (row.Factory || '').toLowerCase() === this.filters.Factory.toLowerCase();

      const matchCategory =
        this.filters.category === 'ALL' ||
        (row.category || '').toLowerCase() === this.filters.category.toLowerCase();

      return matchFormNo && matchEmpId && matchName && matchStatus && matchFactory && matchCategory;
    });

    this.currentPage = 1;
  }

  get pagedProposals(): ProposalRow[] {
    const start = (this.currentPage - 1) * this.pageSize;
    return this.filteredProposals.slice(start, start + this.pageSize);
  }

  get totalPages(): number {
    return Math.max(1, Math.ceil(this.filteredProposals.length / this.pageSize));
  }

  nextPage(): void {
    if (this.currentPage < this.totalPages) {
      this.currentPage++;
    }
  }

  prevPage(): void {
    if (this.currentPage > 1) {
      this.currentPage--;
    }
  }

  search(): void {
    console.log(
      'Search dates:',
      this.formatDate(this.filters.dateFrom),
      this.formatDate(this.filters.dateTo)
    );
    this.applyFilters();
  }

  exportList(): void {
    console.log('Export requested', this.filters);
  }

  goBack(): void {
    this.location.back();
  }

  openFormCode(formCode: string): void {
    console.log('Open proposal detail for', formCode);
  }

  statusClass(status: string): string {
    const v = (status || '').toLowerCase();
    if (v.indexOf('completed') !== -1) return 'status--completed';
    if (v.indexOf('signing') !== -1) return 'status--signing';
    if (v.indexOf('rejected') !== -1) return 'status--rejected';
    if (v.indexOf('pending') !== -1) return 'status--pending';
    return '';
  }
}
