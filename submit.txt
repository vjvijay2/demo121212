-- Package body
 PROCEDURE GET_LOG_REPORT(
    V_SITE IN NUMBER,
    V_VIRTUAL_NUMBER IN VARCHAR2,
    V_COLUMN_NAME IN VARCHAR2,
    V_START_DATE IN VARCHAR2,
    V_END_DATE IN VARCHAR2,
    L_START_PAGE IN VARCHAR2,
    L_PAGE_SIZE IN VARCHAR2,
    V_RES OUT VARCHAR2,
    V_RESULT OUT VARCHAR2,
    V_EMP_ID IN VARCHAR2,
    P_CURSOR OUT SYS_REFCURSOR
) IS
BEGIN
    OPEN P_CURSOR FOR
  
    WITH LICENSEDATA AS(
    
     	SELECT  A.ID,A.SITE,A.VIRTUAL_ID, U.ID AS USER_ID,U.EMP_ID  FROM C_LICENSE_MANAGE_T A  LEFT JOIN  C_USER_INFO_T  U ON A.SITE = U.SITE  WHERE  
     	U.EMP_ID =V_EMP_ID AND   A.DEL_FLAG =0 
     	
     	AND (
--    (TRIM(U.DEPT) = 'All' AND U.DEL_FLAG = 0) 
--    OR 
--    ((TRIM(U.DEPT) <> 'All' AND CAST(A.FIRST_APPLY_DEPART AS VARCHAR(50)) = TRIM(U.DEPT)) AND U.DEL_FLAG = 0) 
--    OR 
--    (A.RESPONSIBLE_DRI_ID = U.ID AND U.DEL_FLAG = 0)
     	  (
    (TRIM(U.DEPT) = 'All' AND U.DEL_FLAG = 0) 
    OR 
    ((TRIM(U.DEPT) <> 'All' AND (CAST(A.FIRST_APPLY_DEPART AS VARCHAR(50)) = TRIM(U.DEPT) OR A.RESPONSIBLE_DEPT_ID = U.DEPT)) AND U.DEL_FLAG = 0) 
    OR 
    (A.RESPONSIBLE_DRI_ID = U.ID AND U.DEL_FLAG = 0)
)
)

    ) 
    SELECT
        RN,
        ROWSCOUNT,
        VIRTUAL_ID,
        SITE_NAME,
        CASE 
            WHEN COLUMN_NAME IN ('FIRST_RESPONSIBLE_DRI', 'RESPONSIBLE_DRI_ID') THEN UI_EMP_ID
            ELSE BEFORE_DATA
        END AS BEFORE_DATA,
        CASE 
            WHEN COLUMN_NAME IN ('FIRST_RESPONSIBLE_DRI', 'RESPONSIBLE_DRI_ID') THEN UD_EMP_ID 
            ELSE AFTER_DATA
        END AS AFTER_DATA,
        BEFORE_DATA_VALUE,
        AFTER_DATA_VALUE,
        COLUMN_NAME,
        RENAMED_COLUMN,
        
        ADD_DATE,
        ADD_BY
    FROM 
        (
        SELECT
            ROW_NUMBER() OVER (ORDER BY A.ADD_DATE DESC) AS RN,
            COUNT(1) OVER () AS ROWSCOUNT,
            D.VIRTUAL_ID,
            CASE
                WHEN D.SITE = 69941 THEN 'FXCN'
                WHEN D.SITE = 95326 THEN 'FXBL'
                ELSE NULL
            END AS SITE_NAME,
            CASE 
                WHEN B.ID IS NOT NULL THEN B.BASIC_CODE 
                ELSE A.BEFORE_DATA
            END AS BEFORE_DATA,
            CASE 
                WHEN C.ID IS NOT NULL THEN C.BASIC_CODE 
                ELSE A.AFTER_DATA
            END AS AFTER_DATA,
            A.COLUMN_NAME,
            A.BEFORE_DATA_VALUE,
            A.AFTER_DATA_VALUE,
            CASE 
                WHEN A.COLUMN_NAME = 'NEW_LICENSE' THEN 'Whether it is a new License?'
                WHEN A.COLUMN_NAME = 'REGION' THEN 'Region Name'
                WHEN A.COLUMN_NAME = 'LICENCE_NO' THEN 'License Number'
                WHEN A.COLUMN_NAME = 'ONE_TIME_CERTIFICATE' THEN 'Whether it is a one-time certification?'
                WHEN A.COLUMN_NAME = 'CERTIFICATE_REQUIRED' THEN 'IF Required'
                WHEN A.COLUMN_NAME = 'VALIDITY_STARTTIME' THEN 'Current License Validity from'
                WHEN A.COLUMN_NAME = 'VALIDITY_ENDTIME' THEN 'Current License Validity To'
                WHEN A.COLUMN_NAME = 'CERTIFICATE_FREQUENCY' THEN 'Certification Frequency (Month)'
                WHEN A.COLUMN_NAME = 'EST_CERTI_DURATION' THEN 'Estimated Certificate Duration(Month)'
                WHEN A.COLUMN_NAME = 'FIRST_RESPONSIBLE_DRI' THEN 'Staff Responsible of certification'
                WHEN A.COLUMN_NAME = 'RESPONSIBLE_DRI_ID' THEN 'Responsible person for the Management'
                WHEN A.COLUMN_NAME = 'ASSIST_DEPT_ID' THEN 'Which dept require assistance in applying License?'
                WHEN A.COLUMN_NAME = 'THIRD_PARTY_AGENCY' THEN 'Have assisted in applying for licenses third party agency/organization'
                WHEN A.COLUMN_NAME = 'ISSUING_INSTITUTION' THEN 'Issuing Department/Institution'
                WHEN A.COLUMN_NAME = 'LICENSE_LEVEL' THEN 'License Level'
                WHEN A.COLUMN_NAME = 'CERTI_LAW_REGULATION' THEN 'Whether it is a certificate required by laws and regulations'
                WHEN A.COLUMN_NAME = 'APPLICATION_FEE' THEN 'Application Fee(INR)'
                WHEN A.COLUMN_NAME = 'AUDIT_REQUIRED' THEN 'Are regular audits required during the validity period of the license?'
                WHEN A.COLUMN_NAME = 'AUDIT_FREQUENCY_MONTH' THEN 'If it will be audited, how often will it be done? (Month)'
                WHEN A.COLUMN_NAME = 'CERTIFICATE_NAME' THEN 'E-Certificate'
                WHEN A.COLUMN_NAME = 'CERTIFICATE_URL' THEN 'E-Certificate URL'
                WHEN A.COLUMN_NAME = 'REMARK' THEN 'Remark'
                ELSE A.COLUMN_NAME
            END AS RENAMED_COLUMN,
          UD.EMP_ID AS UD_EMP_ID,
            UI.EMP_ID AS UI_EMP_ID,
            TO_CHAR(A.ADD_DATE, 'DD-MM-YYYY HH24:MI:SS') AS ADD_DATE,
            A.ADD_BY
        FROM
            LMS.C_LICENSE_LOG_T A
            LEFT JOIN lICENSEDATA D ON A.LICENSE_ID = D.ID
            LEFT JOIN LMS.C_BASE_CODE_MANAGE_T B ON A.COLUMN_NAME IN ('REGION', 'ASSIST_DEPT_ID', 'THIRD_PARTY_AGENCY', 'ISSUING_INSTITUTION', 'LICENSE_LEVEL') 
            AND A.BEFORE_DATA = CAST(B.ID AS VARCHAR2(50))
            LEFT JOIN LMS.C_BASE_CODE_MANAGE_T C ON A.COLUMN_NAME IN ('REGION', 'ASSIST_DEPT_ID', 'THIRD_PARTY_AGENCY', 'ISSUING_INSTITUTION', 'LICENSE_LEVEL') 
            AND A.AFTER_DATA = CAST(C.ID AS VARCHAR2(50))
            LEFT JOIN LMS.C_USER_INFO_T UI ON A.COLUMN_NAME IN ('FIRST_RESPONSIBLE_DRI', 'RESPONSIBLE_DRI_ID') 
            AND A.BEFORE_DATA = CAST(UI.ID AS VARCHAR2(50))
            LEFT JOIN LMS.C_USER_INFO_T UD ON A.COLUMN_NAME IN ('FIRST_RESPONSIBLE_DRI', 'RESPONSIBLE_DRI_ID') 
            AND A.AFTER_DATA = CAST(UD.ID AS VARCHAR2(50))
        WHERE
            D.SITE = NVL(V_SITE, D.SITE) 
            AND D.VIRTUAL_ID = NVL(V_VIRTUAL_NUMBER, D.VIRTUAL_ID) 
            AND A.DEL_FLAG = 0 AND A.COLUMN_NAME = NVL(V_COLUMN_NAME ,A.COLUMN_NAME)
        AND (A.ADD_DATE >= TO_DATE(V_START_DATE, 'DD-MM-YYYY') OR
                     V_START_DATE IS NULL)
         AND (A.ADD_DATE <= TO_DATE(V_END_DATE, 'DD-MM-YYYY') + INTERVAL '1' DAY  OR
                     V_END_DATE IS NULL)
    )-- End date filter
    WHERE
        RN > (L_START_PAGE - 1) * L_PAGE_SIZE
        AND RN <= (L_START_PAGE - 1) * L_PAGE_SIZE + L_PAGE_SIZE;

    V_RES := '1';
    V_RESULT := 'Success';

    COMMIT;
END;

    'RESPONSIBLE_DRI_ID',
    'ASSIST_DEPT_ID',
    'THIRD_PARTY_AGENCY',
    'ISSUING_INSTITUTION',
    'LICENSE_LEVEL',
    'CERTI_LAW_REGULATION',
    'APPLICATION_FEE',
    'AUDIT_REQUIRED',
    'AUDIT_FREQUENCY_MONTH',
    'CERTIFICATE_NAME',
 --   'CERTIFICATE_URL',
    'REMARK'

    V_RESULT := 'Success';

    COMMIT;

 PROCEDURE GET_DEPT_BASE_DASHBOARD_DETAIL (
    V_EMP_ID   IN VARCHAR2,
    V_SITE     IN NUMBER,
    V_DEPT     IN NUMBER,
    V_STATUS   IN NUMBER,
    P_CURSOR   OUT SYS_REFCURSOR
) IS 
BEGIN 
    OPEN P_CURSOR FOR 
    WITH LICENSEDATA AS (
        SELECT 
        A.ID, A.SITE, A.VIRTUAL_ID, U.ID AS USER_ID, U.EMP_ID, 
        A.STATUS, A.RESPONSIBLE_DEPT_ID ,A.DEL_FLAG ,A.FIRST_APPLY_DEPART,U."Role",U.DEPT 
    FROM C_LICENSE_MANAGE_T A  
    LEFT JOIN C_USER_INFO_T U ON A.SITE = U.SITE  
   WHERE U.EMP_ID = V_EMP_ID
    AND A.DEL_FLAG = 0 
    AND A.STATUS != 1201 --ADDED BY DINESH ON 
  AND (
   (TRIM(U.DEPT) = 'All' AND U.DEL_FLAG = 0) 
    OR 
   ((TRIM(U.DEPT) <> 'All' AND (CAST(A.FIRST_APPLY_DEPART AS VARCHAR(50)) = TRIM(U.DEPT) OR A.RESPONSIBLE_DEPT_ID = U.DEPT)) AND U.DEL_FLAG = 0) 
   OR 
    (A.RESPONSIBLE_DRI_ID = U.ID AND U.DEL_FLAG = 0)
)
    )
    SELECT 
        ROW_NUMBER() OVER (ORDER BY B.ID) AS RN, 
        COUNT(*) OVER () AS ROWSCOUNT,
        B.BASIC_CODE AS DEPT, 
        B.ID AS DEPTID,
	  
        A."Role",
        COALESCE(COUNT(A.ID), 0) AS totalcount,
        COALESCE(SUM(CASE WHEN A.STATUS = 4 THEN 1 ELSE 0 END), 0) AS EXPIRED_COUNT,
        COALESCE(SUM(CASE WHEN A.STATUS = 3 THEN 1 ELSE 0 END), 0) AS TO_BE_NEWLY_APPLIED,
        COALESCE(SUM(CASE WHEN A.STATUS = 2 THEN 1 ELSE 0 END), 0) AS NORMAL,
        COALESCE(SUM(CASE WHEN A.STATUS = 5 THEN 1 ELSE 0 END), 0) AS SOON_TO_EXPIRE,
        COALESCE(SUM(CASE WHEN A.STATUS = 6 THEN 1 ELSE 0 END), 0) AS IF_REQUIRED
    FROM C_BASE_CODE_MANAGE_T B 
    LEFT JOIN LICENSEDATA A 
        ON (
            (A.STATUS = 3 AND B.ID = A.FIRST_APPLY_DEPART) 
            OR 
            (A.STATUS <> 3 AND B.ID = A.RESPONSIBLE_DEPT_ID)
        )
    WHERE B.DEL_FLAG = 0 
    AND B.TYPE_ID = 13
    AND A.STATUS = NVL(V_STATUS, A.STATUS)
    AND A.SITE = NVL(V_SITE,A.SITE)
    AND B.ID = NVL(V_DEPT, B.ID)
   AND (
       (EXISTS (SELECT 1 FROM LICENSEDATA WHERE "Role" IN (16, 17))) 
      OR 
       (B.ID = A.DEPT)  -- Restrict to department if role is not 16 or 17
   )
    GROUP BY B.BASIC_CODE, B.ID,A."Role";
END;




 PROCEDURE GET_DASHBOARD_DETAIL_MAIL (
    V_EMP_ID   IN VARCHAR2,
    V_SITE     IN VARCHAR2,
    V_DEPT     IN NUMBER,

    WITH LICENSEDATA AS (
        SELECT 
        A.ID, A.SITE, A.VIRTUAL_ID, U.ID AS USER_ID, U.EMP_ID, 
        A.STATUS, A.RESPONSIBLE_DEPT_ID ,A.DEL_FLAG ,A.FIRST_APPLY_DEPART,U."Role",U.DEPT

        COUNT(*) OVER () AS ROWSCOUNT,
        B.BASIC_CODE AS DEPT, 
        B.ID AS DEPTID,
        A."Role",

    LEFT JOIN LICENSEDATA A

            (A.STATUS = 3 AND B.ID = A.FIRST_APPLY_DEPART) 
            OR 
            (A.STATUS <> 3 AND B.ID = A.RESPONSIBLE_DEPT_ID)

    AND INSTR(',' || V_SITE, ',' || A.SITE || ',') > 0 
--    AND A.SITE = NVL(V_SITE,A.SITE)
    AND B.ID = NVL(V_DEPT, B.ID)
    AND (
        (EXISTS (SELECT 1 FROM LICENSEDATA WHERE "Role" IN (16, 17))) 
        OR 
        (B.ID = A.DEPT)  -- Restrict to department if role is not 16 or 17
    )
    GROUP BY B.BASIC_CODE, B.ID,A."Role";
END;

PROCEDURE GET_STATUS_COUNTS (
    V_EMP_ID IN VARCHAR2,
    V_SITE IN NUMBER,
   -- V_DEPT IN VARCHAR2,
    OUT_CURSOR OUT SYS_REFCURSOR
) 
IS 
BEGIN
    OPEN OUT_CURSOR FOR
    WITH LICENSEDATA AS(
    SELECT A.ID, A.SITE, A.VIRTUAL_ID, U.ID AS USER_ID, U.EMP_ID ,STATUS,A.RESPONSIBLE_DEPT_ID 
    FROM C_LICENSE_MANAGE_T A  
    LEFT JOIN C_USER_INFO_T U ON A.SITE = U.SITE  
    WHERE U.EMP_ID = V_EMP_ID 
    AND A.DEL_FLAG = 0 
    AND (
        TRIM(U.DEPT) = 'All'
        OR 
        (TRIM(U.DEPT) <> 'All' AND CAST(A.FIRST_APPLY_DEPART AS VARCHAR(50)) = TRIM(U.DEPT)
        OR 
        A.RESPONSIBLE_DRI_ID = U.ID
    )
    ))
     
    
    
    
    
    SELECT 'TOTAL' AS types, 'TOTAL' AS type_name, CAST(NULL AS VARCHAR2(50)) AS type_id, COUNT(*) AS counts
FROM LICENSEDATA
UNION ALL
SELECT 'STATUS' AS types, S.BASIC_CODE AS type_name, CAST(A.STATUS AS VARCHAR2(50)) AS type_id, COUNT(A.ID) AS counts
FROM LICENSEDATA A
JOIN C_SP_BASE_CODE_MANAGE_T S ON A.STATUS = S.ID
GROUP BY S.BASIC_CODE, A.STATUS
UNION ALL
